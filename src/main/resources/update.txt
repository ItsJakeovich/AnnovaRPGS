        public void tick(Location sectorLocation, double speed, double vertical) {
                sendPackets(sectorLocation.getWorld(), particles(sectorLocation));
                World world = sectorLocation.getWorld();

                Vector direction = sectorLocation.getDirection().clone().setY(vertical).normalize();
                RayTraceResult result = world.rayTrace(sectorLocation, direction, speed/20, FluidCollisionMode. NEVER,
                        false, 1, null);

                if (result != null) {
                    Material type = result.getHitBlock() == null ? Material.AIR : result.getHitBlock().getType();
                    if ((!type.equals(Material.AIR) && !type.equals(Material.BARRIER) && !type.equals(Material.WATER))
                            || result.getHitEntity() != null) {
                        onHit(sectorLocation);
                        return;
                    }
                }

                sectorLocation.add(direction.multiply(speed/20));

                if (sectorLocation.getBlockY() < world.getMinHeight()) return;

                Bukkit.getScheduler().scheduleSyncDelayedTask(getInstance(), () -> {
                    sendPackets(sectorLocation.getWorld(), particles(sectorLocation));

                    double airResistance = 0.5 * 1.225 * 0.47 * Math.PI * Math.pow(0.022, 2);
                    double drag = airResistance * Math.pow(speed, 2) / 20;

                    double gravity = (0.025 * 9.81) / 20;

                    double newSpeed = Math.max(speed - drag, 0);
                    double newVertical = vertical - gravity;

                    this.tick(sectorLocation, newSpeed, newVertical);
                }, 1L);
            }
